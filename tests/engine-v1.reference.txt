import test from 'node:test';
import assert from 'node:assert/strict';
import {CLASSES,SPELLS,ENCOUNTERS} from '../web/game/data.js';
import {createRoom,addPlayer,applyAction,stats,emptyMeta,item,pack,upgradeMeta} from '../web/game/engine.js';
function battle(n=1,stage=1){let s=createRoom('p0','Tester','guerreiro',emptyMeta(),1234);for(let i=1;i<n;i++)addPlayer(s,'p'+i,'Tester '+i,CLASSES[i%15].id,emptyMeta());s=applyAction(s,'p0',{type:'start'});s.stage=stage;if(stage%5===0){s.stage=stage-1;s.phase='reward';for(let i=0;i<n;i++)s=applyAction(s,'p'+i,{type:'ready'});}else s=applyAction(s,'p0',{type:'route',index:0});return s;}
test('15 classes, 75 paths, 75 finals, 21 unique spells, 18 encounters',()=>{assert.equal(CLASSES.length,15);assert.equal(CLASSES.flatMap(c=>c.paths).length,75);assert.equal(new Set(CLASSES.flatMap(c=>c.paths.map(p=>p.final))).size,75);assert.equal(new Set(SPELLS.map(s=>s.combo)).size,21);assert.equal(ENCOUNTERS.length,18);});
test('party cap and late joining',()=>{let s=battle(20);assert.throws(()=>addPlayer(s,'extra','Extra','mago',emptyMeta()));s=createRoom('p0','Tester','mago',emptyMeta(),1);for(let i=1;i<20;i++)addPlayer(s,'p'+i,'Test','mago',emptyMeta());assert.throws(()=>addPlayer(s,'p20','Test','mago',emptyMeta()),/cheia/);});
test('one action per player per round and immutable input',()=>{let s=battle(2),snapshot=structuredClone(s);s=applyAction(s,'p0',{type:'defend'});assert.deepEqual(snapshot.players[0].acted,false);assert.throws(()=>applyAction(s,'p0',{type:'attack',target:'e0'}),/Aguarde/);assert.equal(s.round,1);s=applyAction(s,'p1',{type:'defend'});assert.equal(s.round,2);assert.equal(s.players[0].acted,false);});
test('boss HP scales and stagger interrupts two enemy actions',()=>{const solo=battle(1,5),party=battle(20,5);assert(party.enemies[0].hp>solo.enemies[0].hp*10);solo.enemies[0].staggerMax=1;let s=applyAction(solo,'p0',{type:'attack',target:'boss'});assert.equal(s.enemies[0].stunned,1);assert.equal(s.players[0].hp,solo.players[0].hp);s=applyAction(s,'p0',{type:'defend'});assert.equal(s.enemies[0].stunned,0);});
test('provocation takes aggro',()=>{let s=battle(2,5);s.players[1].threat=200;s=applyAction(s,'p0',{type:'special'});assert(s.players[0].threat>s.players[1].threat);const hp=s.players[1].hp;s=applyAction(s,'p1',{type:'defend'});assert.equal(s.players[1].hp,hp);});
test('spell costs mana and persists knowledge after defeat',()=>{let s=battle(1,5);s=applyAction(s,'p0',{type:'spell',combo:'IGNIS IGNIS',target:'boss'});assert(s.players[0].meta.spells.includes('IGNIS IGNIS'));assert(s.players[0].mana<stats(s.players[0]).mana);s=applyAction(s,'p0',{type:'abandon'});assert.equal(s.phase,'defeat');assert(s.players[0].meta.spells.includes('IGNIS IGNIS'));assert.equal(s.players[0].meta.runs,1);assert.throws(()=>applyAction(s,'p0',{type:'abandon'}));});
test('all spells resolve and reversed formulas work',()=>{for(const sp of SPELLS){let s=battle(2,5);s.players[0].mana=100;s=applyAction(s,'p0',{type:'spell',combo:sp.combo.split(' ').reverse().join(' '),target:'boss'});assert(s.players[0].meta.spells.includes(sp.combo),sp.name);assert.equal(s.players[0].mana,100-sp.cost);}});
test('all 15 class specials resolve',()=>{for(const c of CLASSES){let s=battle(2,5);s.players[0].classId=c.id;s.players[0].mana=100;s=applyAction(s,'p0',{type:'special',target:'boss'});assert.equal(s.players[0].acted,true,c.name);}});
test('summon limit and expiration',()=>{let s=battle(2,5);s.players[0].classId='necromante';s.players[0].summons=Array.from({length:3},()=>({name:'x',turns:1,power:.1,type:'sombrio'}));assert.throws(()=>applyAction(s,'p0',{type:'special'}),/Limite/);s=applyAction(s,'p0',{type:'defend'});s=applyAction(s,'p1',{type:'defend'});assert.equal(s.players[0].summons.length,0);});
test('skill prerequisites, final evolution and mixed branches',()=>{let s=battle();s.players[0].points=30;s.players[0].level=5;assert.throws(()=>applyAction(s,'p0',{type:'skill',branch:0,node:3}),/anterior/);for(let b=0;b<5;b++)for(let n=0;n<4;n++)s=applyAction(s,'p0',{type:'skill',branch:b,node:n});assert.equal(s.players[0].points,0);assert.equal(Object.keys(s.players[0].skills).length,20);assert.throws(()=>applyAction(s,'p0',{type:'skill',branch:9,node:1}));});
test('loot packs without overlap and oversized inventory fails',()=>{const s=battle(),items=Array.from({length:5},()=>item(s,10)),pos=pack(items);assert(pos);const used=new Set;for(const [i,it] of items.entries())for(let y=pos[i].y;y<pos[i].y+it.h;y++)for(let x=pos[i].x;x<pos[i].x+it.w;x++){const k=y*10+x;assert(!used.has(k));used.add(k);}assert.equal(pack(Array.from({length:61},(_,i)=>({id:i,w:1,h:1}))),null);});
test('invalid actions cannot change original state or create free items',()=>{const s=battle(),snapshot=JSON.stringify(s);assert.throws(()=>applyAction(s,'intruder',{type:'attack'}));assert.throws(()=>applyAction(s,'p0',{type:'equip',item:'fake'}));assert.throws(()=>applyAction(s,'p0',{type:'spell',combo:'BAD',target:'e0'}));assert.equal(JSON.stringify(s),snapshot);});
test('server timeout cannot be triggered early',()=>{const s=battle(2);assert.throws(()=>applyAction(s,'p0',{type:'skipAbsent'},s.roundAt+44000));const n=applyAction(s,'p0',{type:'skipAbsent'},s.roundAt+45000);assert.equal(n.round,2);});
test('permanent upgrade costs and bounds',()=>{const m=emptyMeta();assert.throws(()=>upgradeMeta(m,'vigor'));m.shards=1000;const n=upgradeMeta(m,'vigor');assert.equal(n.shards,980);assert.equal(n.upgrades.vigor,1);assert.equal(m.upgrades.vigor,0);assert.throws(()=>upgradeMeta(m,'other'));});
test('encounter roll is personal and cannot repeat',()=>{let s=battle(2);s.phase='event';s.event=ENCOUNTERS[0];s=applyAction(s,'p0',{type:'encounter'});assert(s.dice.dice>=1&&s.dice.dice<=20);assert.throws(()=>applyAction(s,'p0',{type:'encounter'}));s=applyAction(s,'p1',{type:'encounter',skip:true});assert.equal(s.phase,'reward');});
test('complete 15 stages awards victory exactly once',()=>{let s=createRoom('p0','Test','mago',emptyMeta(),42);s=applyAction(s,'p0',{type:'start'});let loops=0;while(!['victory','defeat'].includes(s.phase)&&loops++<200){if(s.phase==='route')s=applyAction(s,'p0',{type:'route',index:0});else if(s.phase==='combat'){for(const e of s.enemies)e.hp=1;s.players[0].mana=1000;s=applyAction(s,'p0',{type:'special'});}else s=applyAction(s,'p0',{type:'ready'});}assert.equal(s.phase,'victory');assert.equal(s.stage,15);assert.equal(s.players[0].meta.wins,1);assert.equal(s.players[0].meta.shards,143);});
